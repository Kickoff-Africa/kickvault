  search: z.string().max(255).optional(),

});

**credentials.service.ts — MODIFY `listCredentials` signature and body:**

Change the function signature to:
```typescript
export async function listCredentials(userId: string, userRole: string, query?: ListCredentialsQuery)
```

Import `ListCredentialsQuery` from `./credentials.validation.js` and `Prisma` from `@prisma/client`.

Replace the body to build a mutable `where` clause:
```typescript
const isPrivileged = userRole === 'SUPER_ADMIN' || userRole === 'MANAGER';
const where: Prisma.CredentialWhereInput = {
  deletedAt: null,
  ...(!isPrivileged && { credentialAccess: { some: { userId } } }),
};

if (query?.search) {
  where.OR = [
    { friendlyName: { contains: query.search, mode: 'insensitive' } },
    { systemHost: { contains: query.search, mode: 'insensitive' } },
  ];
}

if (query?.expiring_within) {
  const days = parseInt(query.expiring_within.replace('d', ''), 10);
  const cutoff = new Date();
  cutoff.setDate(cutoff.getDate() + days);
  where.expiresAt = { lte: cutoff, gte: new Date() };
}

return prisma.credential.findMany({
  where,
  select: SAFE_CREDENTIAL_SELECT,
  orderBy: { createdAt: 'desc' },
});
```

IMPORTANT: The access control filter (`credentialAccess: { some: { userId } }`) must remain as a top-level where property so Prisma ANDs it with the OR search conditions. Do NOT wrap the OR inside AND — Prisma ANDs all top-level properties automatically.
