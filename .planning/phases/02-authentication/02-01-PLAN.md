    **1. Create src/services/token.service.ts:**

    ```typescript
    import { SignJWT, jwtVerify } from 'jose';
    import crypto from 'node:crypto';
    import { env } from '../config/env.js';

    const secret = new TextEncoder().encode(env.JWT_SECRET);

    export interface AccessTokenPayload {
      sub: string;      // userId
      role: string;
      sessionId: string;
      iss: string;
      iat: number;
      exp: number;
    }

    export async function signAccessToken(payload: {
      userId: string;
      role: string;
      sessionId: string;
    }): Promise<string> {
      return new SignJWT({ role: payload.role, sessionId: payload.sessionId })
        .setProtectedHeader({ alg: 'HS256' })
        .setSubject(payload.userId)
        .setIssuedAt()
        .setExpirationTime(`${env.ACCESS_TOKEN_EXPIRY_MIN}m`)
        .setIssuer('kickvault')
        .sign(secret);
    }

    export async function verifyAccessToken(token: string): Promise<AccessTokenPayload> {
      const { payload } = await jwtVerify(token, secret, {
        issuer: 'kickvault',
        algorithms: ['HS256'],
      });
      return payload as unknown as AccessTokenPayload;
    }

    /**
     * SHA-256 hash any opaque token string before storage or lookup.
     * Used for both refresh tokens and password reset tokens.
     */
    export function hashToken(token: string): string {
      return crypto.createHash('sha256').update(token).digest('hex');
    }

    export function generateRefreshToken(): string {
      return crypto.randomBytes(32).toString('base64url');
    }

    export function generateResetToken(): { plainToken: string; hashedToken: string } {
      const plainToken = crypto.randomBytes(32).toString('base64url');
      const hashedToken = hashToken(plainToken);
      return { plainToken, hashedToken };
    }
    ```

    Note: `generateResetToken` uses `hashToken` internally. The `hashToken` function is also exported so auth.service.ts can hash refresh tokens before storage/lookup without reimplementing SHA-256.

    **2. Create src/services/password.service.ts:**

    ```typescript
    import { hash, verify } from '@node-rs/argon2';

    const HASH_OPTIONS = { memoryCost: 19456, timeCost: 2, parallelism: 1 };
    const TEST_OPTIONS = { memoryCost: 1024, timeCost: 1, parallelism: 1 };

    function getOptions() {
      return process.env.NODE_ENV === 'test' ? TEST_OPTIONS : HASH_OPTIONS;
    }

    export async function hashPassword(password: string): Promise<string> {
      return hash(password, getOptions());
    }

    export async function verifyPassword(hash: string, password: string): Promise<boolean> {
      return verify(hash, password, getOptions());
    }
    ```

    Note: `@node-rs/argon2` verify() returns a boolean (true = match, false = mismatch). It does NOT throw on mismatch.

    **3. Create src/services/email.service.ts:**

    ```typescript
    import nodemailer, { type Transporter } from 'nodemailer';
    import { env } from '../config/env.js';
    import { logger } from '../lib/logger.js';

    let transporter: Transporter | null = null;

    export function setTransporter(mock: Transporter): void {
      transporter = mock;
    }

    export function getTransporter(): Transporter {
      if (!transporter) {
        transporter = nodemailer.createTransport({
          host: env.SMTP_HOST,
          port: env.SMTP_PORT,
          auth: env.SMTP_USER
            ? { user: env.SMTP_USER, pass: env.SMTP_PASS }
            : undefined,
        });
      }
      return transporter;
    }

    export async function sendPasswordResetEmail(
      to: string,
      resetToken: string,
    ): Promise<void> {
      const resetUrl = `${env.FRONTEND_URL}/reset-password?token=${resetToken}`;

      await getTransporter().sendMail({
        from: env.SMTP_FROM,
        to,
        subject: 'Kick Vault - Password Reset',
        text: `You requested a password reset. Click here to reset your password: ${resetUrl}\n\nThis link expires in ${env.RESET_TOKEN_EXPIRY_MIN} minutes.\n\nIf you did not request this, ignore this email.`,
        html: `<p>You requested a password reset.</p><p><a href="${resetUrl}">Click here to reset your password</a></p><p>This link expires in ${env.RESET_TOKEN_EXPIRY_MIN} minutes.</p><p>If you did not request this, ignore this email.</p>`,
      });

      logger.info({ to }, 'Password reset email sent');
    }
    ```

    **4. Create src/middleware/authenticate.ts:**

    ```typescript
    import type { Request, Response, NextFunction } from 'express';
    import { verifyAccessToken } from '../services/token.service.js';
    import { prisma } from '../config/database.js';
    import { UnauthorizedError } from '../lib/errors.js';

    // Track last update time per session to avoid DB writes on every request
    const lastUpdateCache = new Map<string, number>();
    const UPDATE_THRESHOLD_MS = 60_000; // 1 minute

    export async function authenticate(
      req: Request,
      _res: Response,
      next: NextFunction,
    ): Promise<void> {
      const authHeader = req.headers.authorization;

      if (!authHeader?.startsWith('Bearer ')) {
        throw new UnauthorizedError('Missing or invalid authorization header');
      }

      const token = authHeader.slice(7);

      try {
        const payload = await verifyAccessToken(token);

        req.user = { id: payload.sub, role: payload.role };
        req.sessionId = payload.sessionId;

        // Fire-and-forget lastActiveAt update with 1-minute threshold
        const now = Date.now();
        const lastUpdate = lastUpdateCache.get(payload.sessionId) ?? 0;
        if (now - lastUpdate > UPDATE_THRESHOLD_MS) {
          lastUpdateCache.set(payload.sessionId, now);
          prisma.session
            .update({
              where: { id: payload.sessionId },
              data: { lastActiveAt: new Date() },
            })
            .catch(() => {
              // Session may have been deleted (logout) — ignore
              lastUpdateCache.delete(payload.sessionId);
            });
        }

        next();
      } catch (err) {
        if (err instanceof UnauthorizedError) throw err;
        throw new UnauthorizedError('Invalid or expired token');
      }
    }
    ```

    Key design decisions:
    - Does NOT query Session table on every request (access token is stateless)
    - Updates lastActiveAt fire-and-forget with 1-minute threshold to avoid DB pressure
    - Express 5 async error handling: throwing errors works (Express 5 catches async throws)
    - Uses the existing UnauthorizedError from src/lib/errors.ts
